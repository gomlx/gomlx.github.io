// Two-layer MLP on MNIST — full working example
package main

import (
    . "github.com/gomlx/gomlx/core/graph"
    "github.com/gomlx/gomlx/ml/layers"
    "github.com/gomlx/gomlx/ml/model"
)

// modelGraph defines your model as a pure function.
// GoMLX traces it once, compiles to XLA, runs fast.
func modelGraph(scope *model.Scope, inputs []*Node) *Node {
    x := inputs[0]                                     // [batch, 784]
    x  = layers.Dense(scope.In("hidden"), x, true, 128) // → [batch, 128]
    x  = Relu(x)
    x  = layers.Dense(scope.In("output"), x, true, 10)  // → [batch, 10]
    return x
}
// Build a trainer with loss, optimizer, and metrics
trainer := train.NewTrainer(backend, store, modelGraph,
    loss.SparseCategoricalCrossentropyLogits,
    optimizer.Adam().LearningRate(0.001).Done(),
    []metric.Interface{metric.NewMovingAverageSparseCategoricalLogitsAccuracy("acc", "acc")},
    []metric.Interface{metric.NewMeanSparseCategoricalLogitsAccuracy("acc", "acc")},
)

// Loop runs entirely on-device — no Python round-trips
loop := train.NewLoop(trainer)
commandline.AttachProgressBar(loop) // Dynamic progress bar in CLI
loop.RunSteps(trainDS, 10_000)

// Evaluate on test set
metrics, _ := trainer.Eval(testDS)
fmt.Printf("Test accuracy: %s\n", metrics[0].PrettyPrint())
